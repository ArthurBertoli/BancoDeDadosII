--1)Gatilho para impedir a inserção ou atualização de Clientes com o mesmo CPF.
--onde? Clientes                    }
--quando? Antes                     } Elementos para ajudar na 
--nível? row level                  } criação do gatilho
--operações? inserir ou atualizar   }
create or replace function valida_clientes() returns trigger as
$$
begin
	if TG_OP = 'INSERT' then
		if (select 1 from cliente where cpf = new.cpf) then
			raise exception 'Já existe um cliente com ese cpf';
		end if;
	elsif TG_OP = 'UPDATE' then
		if(select 1 from cliente where cpf = new.cpf) then
			raise exception 'Já existe um xliente com esse cpf';
		end if;
	end if;
	return new;
end;
$$
language plpgsql;

create trigger valida_clientes before insert or update on cliente for each row execute procedure valida_clientes()

--5)Gatilho para impedir que um mecânico seja removido caso não exista outro mecânico com a mesma função.
--onde: tabela mecanico
--quando: before
--operações: update/delete
--nivel: row level 
create or replace function mec_funcao() returns trigger as
$$
begin
	if (TG_OP = 'DELETE') then
		if ((select count (*) from mecanico where funcao ilike old.funcao) <= 1) then
			raise exception 'Proibido excluir mecanico';
		end if;
		return old;
	end if;
	if (TG_OP = 'UPDATE') then
		if ((select count (*) from mecanico where funcao ilike old.funcao) <= 1 and new.funcao <> old.funcao) then
			raise exception 'Proibido atualizar mecanico';
		end if;
		return new;
	end if;
end;
$$
language plpgsql;

create trigger mec_funcao before update or delete on mecanico for each row execute procedure mec_funcao()

--8)Gatilho para impedir que mais de 1 conserto seja agendado no mesmo setor na mesma hora.
create or replace function impedir_agendamento() returns trigger as 
$$
begin
	if (TG_OP = 'INSERT') then
		if (select 1 from conserto c join mecanico m using (codm) join setor s using (cods) where c.data = new.data and s.cods in (new.cods)) then
			raise exception 'Setor já alocado para esta data';
		end if;
	elsif (TG_OP = 'UPDATE') then
		if (select 1 from conserto c join mecanico m using (codm) join setor s using (cods) where c.data = new.data and s.cods in (new.cods) and new.hora = old.hora) then
			raise exception 'Setor já alocado para esta data';
		end if;
	end if;
	return new;
end;
$$
language plpgsql;
