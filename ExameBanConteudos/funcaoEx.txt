--2)Função para exclusão de um mecânico. 
CREATE OR REPLACE FUNCTION remove_mecanico (pcodm int)
returns int AS
$$
	declare
		linhas int default 0;
	BEGIN
		delete from mecanico where codm = pcodm;
		get diagnostics linhas = ROW_COUNT;
		return linhas;
	END;
$$
language plpgsql;

--3)Função única para inserção, atualizar e exclusão de um cliente.
CREATE OR REPLACE FUNCTION atualiza_clientes (op varchar, pcodc int, pcpf varchar, pnome varchar, pidade int, pendereco varchar, pcidade varchar)
returns void AS
$$
	BEGIN
		if upper(op) = 'INSERT' then
			insert into cliente values (pcodc, pcpf, pnome, pidade, pendereco, pcidade);
		elseif upper(op) = 'UPDATE' then
			update cliente set
				cpf = pcfp,
				nome = pnome,
				idade = pidade,
				endereco = pendereco,
				cidade = pcidade
			where
				codc = pcodc;
		elseif upper(op) = 'DELETE' then 
			delete from cliente where codc = pcodc;
		else
			raise exception 'Comando inválido, apenas INSERT, UPDATE ou DELETE';
		end if;
		return;

	END;
$$
language plpgsql;

select atualiza_clientes ('update', 5, '51000110000', 'Carlos', 44, 'Centro', 'Florianópolis'); --Errada

--4)Função que limite o cadastro de no máximo 10 setores na oficina mecânica.
CREATE OR REPLACE FUNCTION cadastra_setores(pcods int, pnome varchar)
returns void AS
$$
declare
	quant int default 0;
	BEGIN
		select count(*) into quant from setor;
		if quant < 10 then
			insert into setor values(pcods, pnome);
		else
			raise exception 'Limite de setores já atingido.';
		end if;
		return;
	END;
$$
language plpgsql

--5)Função que limita o cadastro de um conserto apenas se o mecânico não tiver mais de 3 consertos agendados para o mesmo dia.
CREATE OR REPLACE FUNCTION cadastro_consertos (pcodm int, pcodv int, pdata date, phora time)
returns void AS
$$
declare
	quantmec int default 0;
	BEGIN
		select count(*) from conserto group by codm, data into quantmec;
		raise notice 'Valor de quant: %', quantmec;
		
		if quantmec < 3 then
			 insert into conserto values (pcodm, pcodv, pdata, phora);
		else 
			raise exception 'Quantidade máxima de consertos já atingida.';
		end if;
		return;
	END;
$$
language plpgsql;

--6)Função para calcular a média geral de idade dos Mecânicos e Clientes.
CREATE OR REPLACE FUNCTION media() returns float as
$$
declare 
	soma float default 0;
	v_idade float default 0;
	quant float default 0;
BEGIN
	for v_idade in select idade from mecanico where idade is not null loop
		soma := soma + v_idade;
		quant := quant + 1;
	end loop;
	for v_idade in select idade from cliente where idade is not null loop
		soma := soma + v_idade;
		quant := quant + 1;
	end loop;
	return ROUND ((soma / quant)::numeric, 2 );
END;
$$
language plpgsql;

--7)Função única que permita fazer a exclusão de um Setor, Mecânico, Cliente ou Veículo.
CREATE OR REPLACE FUNCTION exclusao(tabela varchar, id varchar, valor int) returns int as
$$
declare
	sql varchar;
	linhas_afetadas int;
BEGIN
	sql := 'delete from ' ||tabela|| ' where '||id||' = '||valor;
	raise notice 'SQL = %', sql;
	execute sql;
	get diagnostics linhas_afetadas = ROW_COUNT;
	return linhas_afetadas;
END;
$$
language plpgsql;
select exclusao ('setor', 'cods', 1);

--que possuir mais consertos agendados, etc. Para testar a função, não se esqueça de inserir na tabela alguns clientes com este problema.
CREATE OR REPLACE FUNCTION exclui_cpf() returns int as 
$$
declare
	vcodc int;
	vcpf char(11): totalLinhas int := 0;
	cont int := 0; linhas int := 0;
BEGIN
	for vcpf in select cpf from cliente group by cpf having count(*)>1 loop
	cont := 0;
		for vcodc from cliente c where c.cpf =vcpf
			if cont > 0 then
			delete from cliente cli where cli.codc = v.codc;
			get diagnostics linhas = row_count;
			totalLinhas := totalLinhas + linhas;
			end if;
			cont := cont + 1;
		end loop;
	end loop;
	return totalLinhas;
END;
$$
language plpgsql;
